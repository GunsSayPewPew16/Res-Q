def run_matching_pipeline(surplus_list, shelters):
    valid_pairs = []
    for item in surplus_list:
        for shelter in shelters:
            if hard_filter(item, shelter):
                score = calculate_match_score(item, shelter)
                valid_pairs.append((score, item, shelter))
    return optimize_global_allocation(valid_pairs)
